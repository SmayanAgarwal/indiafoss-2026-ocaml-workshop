% A resource has a lifecycle (acquire, use, release); three bugs are
% the three ways to get the release wrong. Used in M10-L04.
\documentclass[tikz,border=4pt]{standalone}
\usepackage{tikz}
\input{_m10-styles.texinput}
\begin{document}
\begin{tikzpicture}[m10, x=1mm, y=-1mm, font=\large]
  % the good lifecycle
  \node[safebox, minimum width=30mm, minimum height=13mm, font=\large] (acq) at (16,0) {acquire};
  \node[safebox, minimum width=30mm, minimum height=13mm, font=\large] (use) at (62,0) {use};
  \node[safebox, minimum width=30mm, minimum height=13mm, font=\large] (rel) at (108,0) {release};
  \draw[goodflow, line width=1pt] (acq) -- (use);
  \draw[goodflow, line width=1pt] (use) -- (rel);

  % three failure modes
  \node[freedcell, minimum width=34mm, minimum height=18mm, align=center, font=\large] (leak) at (18,30)
    {\textbf{leak}\\never release};
  \node[freedcell, minimum width=34mm, minimum height=18mm, align=center, font=\large] (dbl) at (66,30)
    {\textbf{double-close}\\release twice};
  \node[freedcell, minimum width=40mm, minimum height=18mm, align=center, font=\large] (uac) at (116,30)
    {\textbf{use-after-close}\\use past release};

  \draw[badflow, line width=1pt] (rel.south) .. controls +(0,7) and +(0,-7) .. (leak.north);
  \draw[badflow, line width=1pt] (rel.south) -- (dbl.north);
  \draw[badflow, line width=1pt] (rel.south) .. controls +(0,7) and +(0,-7) .. (uac.north);

  \node[note, anchor=north west, font=\large\itshape] at (-2,44)
    {The memory bug zoo, one level up: leak \(\approx\) never free,};
  \node[note, anchor=north west, font=\large\itshape] at (-2,54)
    {double-close \(\approx\) double-free, use-after-close \(\approx\) use-after-free.};
\end{tikzpicture}
\end{document}
