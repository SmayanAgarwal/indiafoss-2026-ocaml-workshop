% Why a handle that escapes its with_* scope is already closed.
% Used in M10-L04.
\documentclass[tikz,border=4pt]{standalone}
\usepackage{tikz}
\input{_m10-styles.texinput}
\begin{document}
\begin{tikzpicture}[m10, x=1mm, y=-1mm, font=\large]
  % the scope box
  \node[safebox, minimum width=72mm, minimum height=30mm, anchor=north west] (scope) at (0,4) {};
  \node[goodnote, anchor=north west, font=\large\bfseries] at (3,5)
    {with\_open\_socket (fun s $\to$};
  \node[goodnote, anchor=south west, font=\large\bfseries] at (3,33)
    {) close fires here, always};

  % the handle inside
  \node[heapcell, minimum width=22mm, minimum height=12mm, font=\large] (h) at (26,19) {socket \texttt{s}};

  % the handle escaping
  \node[freedcell, hatched, minimum width=24mm, minimum height=12mm, font=\large] (esc) at (104,19)
    {\texttt{s} (dead)};
  \draw[badflow, line width=1pt] (h.east) -- node[badnote, above, font=\large]{returned out} (esc.west);

  \node[badnote, anchor=north west, font=\large] at (0,40)
    {The escaped handle is already closed: using it is use-after-close,};
  \node[badnote, anchor=north west, font=\large] at (0,50)
    {and scoping cannot stop it.};
\end{tikzpicture}
\end{document}
