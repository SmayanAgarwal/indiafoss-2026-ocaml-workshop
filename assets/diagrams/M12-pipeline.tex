\begin{document}
\begin{tikzpicture}[m10, x=1mm, y=-1mm,
    arr/.style={-{Stealth[length=2.5mm]}, draw=labelgrey,
                line width=0.8pt},
    arrlbl/.style={note, font=\sffamily\small, fill=white,
                   inner sep=1.5pt}]
  % Sources you write (green)
  \node[safebox, minimum width=28mm, minimum height=11mm,
        font=\sffamily\large, anchor=north west] (config) at (0,8)
        {config.ml};
  \node[safebox, minimum width=28mm, minimum height=11mm,
        font=\sffamily\large, anchor=north west] (uni) at (0,52)
        {unikernel.ml};
  % Generated build context (amber)
  \node[warnbox, minimum width=42mm, minimum height=9mm,
        font=\sffamily, anchor=north west] (mk) at (52,0)
        {Makefile, dune files};
  \node[warnbox, minimum width=42mm, minimum height=9mm,
        font=\sffamily, anchor=north west] (mainml) at (52,12)
        {mirage/main.ml};
  \node[warnbox, minimum width=42mm, minimum height=9mm,
        font=\sffamily, anchor=north west] (opam) at (52,24)
        {mirage/hello-hvt.opam};

  \node[greybox, minimum width=42mm, minimum height=9mm,
        font=\sffamily, anchor=north west] (deps) at (52,44)
        {duniverse/ (packages)};
  % Output
  \node[box, fill=stackfill, draw=stackedge, text=stackedge,
        minimum width=26mm, minimum height=16mm,
        font=\sffamily\large\bfseries, anchor=north west]
        (img) at (118,26) {image\\(ELF)};
  % Arrows
  \draw[arr] (config.east) -- (50,13.5)
        node[arrlbl, pos=0.5, above] {mirage configure};
  \draw[arr] (opam.south) -- (deps.north)
        node[arrlbl, pos=0.5, right=2mm] {make depend};
  \draw[arr] (mainml.east) -- (img.north west);
  \draw[arr] (deps.east) -- (img.west);
  \draw[arr] (uni.east) to[out=0,in=-90]
        node[arrlbl, pos=0.3, below=1.5mm] {dune build} (img.south);
\end{tikzpicture}
