% Memory-safety bugs as a share of severe bugs, four vendor studies.
% Horizontal bars. Used in M10-L01 and M12-L01. Fonts are sans and
% oversized relative to the canvas so they survive being scaled
% into a slide column.
\documentclass[tikz,border=4pt]{standalone}
\usepackage{tikz}
\input{_m10-styles.texinput}
\begin{document}
\begin{tikzpicture}[m10, x=1mm, y=-1mm, font=\sffamily]
  \def\barmax{90}
  \foreach \y/\pct/\lbl in {
      0/70/{Microsoft (2019)},
      18/70/{Chrome (2020)},
      36/86/{Android critical (2022)},
      54/67/{0-days in the wild (2021)}} {
    \node[note, anchor=east, font=\sffamily\Large] at (-4,\y+7) {\lbl};
    \draw[draw=labelgrey!30, line width=0.4pt, fill=white]
      (0,\y) rectangle (\barmax,\y+14);
    \fill[secfill, draw=secedge, line width=0.6pt]
      (0,\y) rectangle ({\pct/100*\barmax},\y+14);
    \node[note, anchor=west, font=\sffamily\huge\bfseries,
          text=secedge!75!black]
      at ({\pct/100*\barmax + 3},\y+7) {\pct\%};
  }
  \node[note, anchor=north west, font=\sffamily\Large\itshape]
    at (-52,78)
    {Share of severe bugs that are memory-safety bugs.};
  \node[note, anchor=north west, font=\sffamily\Large\itshape]
    at (-52,89)
    {Microsoft's share was flat for over a decade.};
\end{tikzpicture}
\end{document}
